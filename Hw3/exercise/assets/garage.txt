# --- VARIABLES ---
# Define dimensions (Values from your C++ code)
THICKNESS = 0.25
WIDTH = 7.3152
LENGTH = 3.8608
HEIGHT = 4.0

# --- GARAGE ROOT ---
# This initializes the anchor position
OBJECT 0 0 0
BEGIN
    # --- RIGHT WALL ---
    # CUBE x y z w h l
    CUBE 0 0 0 THICKNESS HEIGHT WIDTH
    # TEXTURE WOOD

    # --- BACK/SIDE WALL ---
    # Position X: -(LENGTH + THICKNESS) / 2
    # Position Z: -(WIDTH - THICKNESS) / 2
    # We use (0 - (...)) to safely handle negative math in the simple parser
    CUBE (0-(LENGTH+THICKNESS))/2 0 (0-(WIDTH-THICKNESS))/2 THICKNESS HEIGHT LENGTH
    
    # Rotate 90 degrees around Y-axis (0, 1, 0)
    ROTATION 90 0 1 0
    # TEXTURE WOOD
END
